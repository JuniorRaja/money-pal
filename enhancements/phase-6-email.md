### Weekly Report Content

```typescript
function formatWeeklyReportHtml(data: WeeklyReportData): string {
  return `
    <h2>Your Week in Review</h2>
    <p>${data.transactionCount} transactions totaling ${formatMoney(data.totalSpend)}</p>
    
    <h3>Budget Pace</h3>
    ${data.budgetPace.map(b => `
      <p>${b.category}: ${b.percentUsed}% used, ${b.daysRemaining} days left</p>
    `).join('')}
    
    <h3>Upcoming</h3>
    ${data.upcomingBills.map(b => `
      <p>${b.name}: ${formatMoney(b.amount)} due ${b.dueDate}</p>
    `).join('')}
  `;
}
```
